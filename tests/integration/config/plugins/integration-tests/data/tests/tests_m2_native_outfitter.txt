test "M2 Native Single Ship Routes"
	status active
	sequence
		inject "M2 Native Single Ship Save"
		call "Load First Savegame"
		assert
			"credits" == 100000000
			"outfit (flagship installed): M2 Test Module" == 0
			"outfit (cargo): M2 Test Module" == 0
		input
			key o
		input
			outfit "M2 Test Module"
		input
			key b
		assert
			"credits" == 99999000
			"outfit (flagship installed): M2 Test Module" == 1
			"outfit (cargo): M2 Test Module" == 0
		# Ship -> cargo, then cargo -> storage and storage -> cargo.
		input
			key u
		assert
			"credits" == 99999000
			"outfit (flagship installed): M2 Test Module" == 0
			"outfit (cargo): M2 Test Module" == 1
			"outfit (storage): M2 Test Module" == 0
		input
			key r
		assert
			"outfit (cargo): M2 Test Module" == 0
			"outfit (storage): M2 Test Module" == 1
		input
			key c
		assert
			"outfit (cargo): M2 Test Module" == 1
			"outfit (storage): M2 Test Module" == 0
		# Cargo -> ship; selling falls through empty cargo/storage to ship -> shop.
		input
			key i
		assert
			"outfit (flagship installed): M2 Test Module" == 1
			"outfit (cargo): M2 Test Module" == 0
		input
			key s
		assert
			"credits" == 100000000
			"outfit (flagship installed): M2 Test Module" == 0
		# Empty storage falls through to shop -> cargo, then cargo -> shop.
		input
			key c
		assert
			"credits" == 99999000
			"outfit (cargo): M2 Test Module" == 1
		input
			key s
		assert
			"credits" == 100000000
			"outfit (cargo): M2 Test Module" == 0
		# Empty cargo/ships fall through to shop -> storage. Empty cargo then
		# falls through to storage -> ship. Finally ship -> storage, then storage -> shop.
		input
			key r
		assert
			"credits" == 99999000
			"outfit (storage): M2 Test Module" == 1
		input
			key i
		assert
			"outfit (storage): M2 Test Module" == 0
			"outfit (flagship installed): M2 Test Module" == 1
		input
			key r
		assert
			"outfit (flagship installed): M2 Test Module" == 0
			"outfit (storage): M2 Test Module" == 1
		input
			key s
		assert
			"credits" == 100000000
			"outfit (storage): M2 Test Module" == 0
		input
			key escape

		assert

			"outfit (storage): M2 Test Module" == 1
		input
			key c
		assert
			"outfit (cargo): M2 Test Module" == 1
			"outfit (storage): M2 Test Module" == 0
		# Cargo -> ship; selling falls through empty cargo/storage to ship -> shop.
		input
			key i
		assert
			"outfit (flagship installed): M2 Test Module" == 1
			"outfit (cargo): M2 Test Module" == 0
		input
			key s
		assert
			"credits" == 100000000
			"outfit (flagship installed): M2 Test Module" == 0
		# Empty storage falls through to shop -> cargo, then cargo -> shop.
		input
			key c
		assert
			"credits" == 99999000
			"outfit (cargo): M2 Test Module" == 1
		input
			key s
		assert
			"credits" == 100000000
			"outfit (cargo): M2 Test Module" == 0
		# Empty cargo/ships fall through to shop -> storage. Empty cargo then
		# falls through to storage -> ship. Finally ship -> storage, then storage -> shop.
		input
			key r
		assert
			"credits" == 99999000
			"outfit (storage): M2 Test Module" == 1
		input
			key i
		assert
			"outfit (storage): M2 Test Module" == 0
			"outfit (flagship installed): M2 Test Module" == 1
		input
			key r
		assert
			"outfit (flagship installed): M2 Test Module" == 0
			"outfit (storage): M2 Test Module" == 1
		input

			key escape
